SELECT CAST(C.IDCLIENTE AS int) AS IDCLIENTE,
       CAST(LTRIM(RTRIM(CONCAT(C.NOMBRE, ' ', C.APELLIDO))) AS varchar(100)) AS NOMBRE,
       CAST(CASE C.SEXO
            WHEN 'M' THEN 'Masculino'
            WHEN 'F' THEN 'Feminino'
            ELSE C.SEXO
       END AS varchar(20)) AS SEXO,
       CAST(LTRIM(RTRIM(C.EMAIL)) AS varchar(60)) AS EMAIL,
       CAST(C.FECHA_NACIMIENTO AS date) AS FECHA_NACIMIENTO,
       CAST(D.CIUDAD AS varchar(100)) AS CIUDAD,
       CAST(D.ESTADO AS varchar(30)) AS ESTADO,
       CAST(D.REGION AS varchar(30)) AS REGION
FROM dbo.CLIENTE AS C
LEFT JOIN dbo.DIRECCION AS D
  ON D.ID_CLIENTE = C.IDCLIENTE;
