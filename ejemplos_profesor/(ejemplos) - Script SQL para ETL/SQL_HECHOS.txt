SELECT CAST(FAC.IDFACTURA AS int) AS IDFACTURA,
       CAST(FAC.ID_CLIENTE AS int) AS IDCLIENTE,
       CAST(FAC.ID_VENDEDOR AS int) AS IDVENDEDOR,
       CAST(FAC.ID_FORMA AS int) AS IDFORMA,
       CAST(I.ID_PRODUCTO AS int) AS IDPRODUCTO,
       CAST(P.ID_PROVEEDOR AS int) AS IDPROVEEDOR,
       CAST(FAC.FECHA AS date) AS FECHA,
       CAST(I.CANTIDAD AS int) AS CANTIDAD,
       CAST(I.TOTAL AS decimal(10,2)) AS TOTAL_ITEM,
       CAST(I.CANTIDAD * P.COSTO_MEDIO AS decimal(10,2)) AS COSTO_TOTAL,
       CAST(I.TOTAL - (I.CANTIDAD * P.COSTO_MEDIO) AS decimal(10,2)) AS UTILIDAD_TOTAL
FROM dbo.FACTURA AS FAC
INNER JOIN dbo.ITEM_FACTURA AS I
  ON I.ID_FACTURA = FAC.IDFACTURA
INNER JOIN dbo.PRODUCTO AS P
  ON P.IDPRODUCTO = I.ID_PRODUCTO;
